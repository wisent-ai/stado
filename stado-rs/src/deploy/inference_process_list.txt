set -euo pipefail
if ! command -v nvidia-smi >/dev/null; then
  printf 'ERROR\tnvidia-smi missing\n'
  exit 1
fi
nvidia-smi --query-compute-apps=pid,used_memory --format=csv,noheader,nounits |
while IFS=, read -r raw_pid raw_used; do
  pid="${raw_pid//[[:space:]]/}"
  used_mib="${raw_used//[[:space:]]/}"
  case "$pid" in ''|*[!0-9]*) continue ;; esac
  [ -r "/proc/$pid/stat" ] || continue
  stat_tail=$(sed 's/^.*) //' "/proc/$pid/stat") || continue
  set -- $stat_tail
  start_ticks="${20:-}"
  case "$start_ticks" in ''|*[!0-9]*) continue ;; esac
  owner=$(stat -c '%U' "/proc/$pid" 2>/dev/null || printf unknown)
  command=$(tr '\000\t\r\n' '    ' < "/proc/$pid/cmdline" 2>/dev/null || true)
  [ -n "$command" ] || command=$(cat "/proc/$pid/comm" 2>/dev/null || printf unknown)
  cgroup=$(tr '\t\r\n' '   ' < "/proc/$pid/cgroup" 2>/dev/null || true)
  printf 'PROCESS\t%s:%s\t%s\t%s\t%s\t%s\t%s\t%s\n' \
    "$pid" "$start_ticks" "$pid" "$start_ticks" "$used_mib" "$owner" "$command" "$cgroup"
done
